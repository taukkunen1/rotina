0;

    let progressMsg;
    if(applicable.length === 0){ progressMsg = 'nada marcado ainda'; }
    else if(pct === 100){ progressMsg = 'tudo certo por aqui! ✅'; }
    else if(remaining === 1){ progressMsg = 'falta só 1! 🔥'; }
    else { progressMsg = `${doneCount}/${applicable.length} feitas`; }

    const ul = document.createElement('ul');
    ul.className = 'tasks';
    const nextSuggestedId = period.tasks.find(t => !t.external && !state.checkedToday[t.id])?.id;
    period.tasks.forEach(task=>{
      const li = document.createElement('li');
      const status = state.checkedToday[task.id]; // 'done' | 'na' | 'x' | undefined
      const isSuggested = task.id === nextSuggestedId;
      const late = status === 'done' && isTaskLate(task, period.time);
      li.className = 'task'
        + (status === 'done' ? ' done' : '')
        + (status === 'na' ? ' na' : '')
        + (status === 'x' ? ' notdone' : '')
        + (isSuggested ? ' suggested' : '')
        + (late ? ' late-done' : '');
      li.innerHTML = `
        <span class="mark-group" data-id="${task.id}">
          <button type="button" class="mark-btn mark-done ${status==='done'?'active':''}" data-status="done" title="Feito">✓</button>
          <button type="button" class="mark-btn mark-na ${status==='na'?'active':''}" data-status="na" title="Não precisou realizar hoje">–</button>
          <button type="button" class="mark-btn mark-x ${status==='x'?'active':''}" data-status="x" title="Não feito, perde metade dos pontos">✕</button>
        </span>
        <span class="txt">${isSuggested ? '<span class="suggested-flag">👉</span>' : ''}<span class="task-icon">${taskIcon(task.txt)}</span>${task.txt}${task.sub ? `<span class="sub">${task.sub}</span>` : ''}${late ? '<span class="late-flag">⏰ atrasado</span>' : ''}</span>
        <span class="pts">${status==='na' ? 'n/a' : status==='x' ? '−'+taskNotDonePenalty(task) : '+'+task.pts}</span>
      `;
      ul.appendChild(li);
    });

    div.innerHTML = `
      <div class="period-head"><h2>${period.label}</h2><span class="time">${period.time}</span></div>
      <span class="time-remaining" id="timeRemaining_${key}"></span>
      <span class="progress-txt">${progressMsg}</span>
      <div class="progress-bar"><div style="width:${pct}%"></div></div>
    `;
    div.dataset.periodTime = period.time;
    div.appendChild(ul);
    periodsEl.appendChild(div);
  });

  updateTimeRemainingDisplays();
  startTimeRemainingTicking();

  periodsEl.querySelectorAll('.mark-btn').forEach(btn=>{
    btn.addEventListener('click', onMarkClick);
  });

  const badEl = document.getElementById('badHabits');
  badEl.innerHTML = '';
  CONFIG.badHabits.forEach(h=>{
    const isPositive = h.pts >= 0;
    const btn = document.createElement('button');
    btn.className = 'bad-btn' + (isPositive ? ' positive' : '');
    btn.innerHTML = `${h.txt}<span class="minus">${isPositive?'+':'−'}${Math.abs(h.pts)}</span>`;
    btn.addEventListener('click', ()=>{
      const delta = addPoints(h.pts);
      addLog(`${isPositive?'+':'−'}${Math.abs(h.pts)} · ${h.txt}`);
      saveState(state);
      bumpTotal(delta, btn);
      renderLog();
    });
    badEl.appendChild(btn);
  });

  renderRewards();
  renderLog();
  renderHistory();
  renderAchievementMural();
  renderWeeklyReview();
  renderBackupReminder();
  renderTestPanel();
  renderTestModeBanner();
  renderDriveSync();
}

function daysBetween(isoA, isoB){
  const a = new Date(isoA+"T00:00:00");
  const b = new Date(isoB+"T00:00:00");
  return Math.round((b-a)/86400000);
}

function shiftISODate(iso, deltaDays){
  const d = new Date(iso+"T00:00:00");
  d.setDate(d.getDate()+deltaDays);
  return d.getFullYear()+"-"+String(d.getMonth()+1).padStart(2,'0')+"-"+String(d.getDate()).padStart(2,'0');
}
