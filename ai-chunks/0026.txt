
<script>
/* ===================== REVISION CONTROL + STEP 3 CONFLICT MERGE ===================== */
(function(){
  const BASELINE_KEY = 'hector_rotina_drive_baseline_v1';
  const MAX_RETRIES = 2;
  const clone = v => v == null ? v : JSON.parse(JSON.stringify(v));
  const equal = (a,b) => JSON.stringify(a) === JSON.stringify(b);
  const isObject = v => v && typeof v === 'object' && !Array.isArray(v);
  function getBaseline(){ try{return JSON.parse(localStorage.getItem(BASELINE_KEY)||'null');}catch(e){return null;} }
  function setBaseline(config, appState, revision){ try{localStorage.setItem(BASELINE_KEY, JSON.stringify({config:clone(config),state:clone(appState),revision:Number(revision||0)}));}catch(e){} }

  function mergeValue(base, local, remote){
    if(equal(local,base)) return clone(remote);
    if(equal(remote,base)) return clone(local);
    if(equal(local,remote)) return clone(local);
    if(typeof local === 'boolean' && typeof remote === 'boolean') return local || remote;
    if(Array.isArray(local) && Array.isArray(remote)){
      const out=[], n=Math.max(local.length,remote.length,Array.isArray(base)?base.length:0);
      for(let i=0;i<n;i++) out[i]=mergeValue(Array.isArray(base)?base[i]:undefined,local[i],remote[i]);
      return out;
    }
    if(isObject(local) && isObject(remote)){
      const out={}, keys=new Set([...Object.keys(isObject(base)?base:{}),...Object.keys(local),...Object.keys(remote)]);
      keys.forEach(k=>out[k]=mergeValue(isObject(base)?base[k]:undefined,local[k],remote[k]));
      return out;
    }
    return clone(remote); // conflicting scalar edits: newer remote value wins conservatively
  }

  async function postBackup(payload){
    const res=await fetch(DATA_ENDPOINT,{method:'POST',headers:{'Content-Type':'text/plain;charset=utf-8'},body:JSON.stringify(payload)});
    if(!res.ok) throw new Error('Falha ao salvar');
    return res.json();
  }

  window.driveSave=async function(options){
    options=options||{}; const silent=!!options.silent; let attempts=0;
    try{
      while(attempts<=MAX_RETRIES){
        const localConfig=clone(CONFIG), localState=clone(state), base=getBaseline();
        const baseRevision=Number((base&&base.revision)||state.driveRevision||0);
        const data=await postBackup({config:localConfig,state:localState,baseRevision:baseRevision,exportedAt:new Date().toISOString()});
        if(data&&data.ok){
          const revision=Number(data.revision||baseRevision+1);
          state.driveRevision=revision; state.driveLastSyncISO=todayISO();
          saveState(state); setBaseline(CONFIG,state,revision);
          if(typeof renderDriveSync==='function') renderDriveSync();
          if(!silent) alert(attempts?'Conflito resolvido e alterações sincronizadas!':'Salvo no Google Drive!');
          return {ok:true,revision,merged:attempts>0};
        }
        if(!(data&&data.conflict)) throw new Error((data&&data.error)||'Falha ao salvar');
        if(attempts>=MAX_RETRIES) throw new Error('Conflito persistente');
        const remote=data.data||{}, remoteRevision=Number(data.revision||remote.revision||0);
        const baseline=base||{config:{},state:{},revision:baseRevision};
        CONFIG=mergeValue(baseline.config||{},localConfig||{},remote.config||{});
        state=mergeValue(baseline.state||{},localState||{},remote.state||{});
        state.driveRevision=remoteRevision;
        saveConfig(CONFIG); saveState(state); setBaseline(remote.config||{},remote.state||{},remoteRevision);
        if(typeof render==='function') render();
        attempts++;
      }
    }catch(err){
      console.error('Drive sync error',err);
      if(!silent) alert('Não consegui sincronizar agora. As alterações locais foram preservadas para nova tentativa.');
      return {ok:false,error:err};
    }
  };

  window.driveLoad=async function(){
    try{
      const res=await fetch(DATA_ENDPOINT,{method:'GET',cache:'no-store'});
      if(!res.ok) throw new Error('Falha ao carregar');
      const backup=await res.json();
      if(backup.error==='not_found'){alert('Nenhum backup encontrado no Google Drive ainda.');return;}
      if(!confirm('Isso vai substituir os dados atuais pelos que estão salvos no Google Drive. Continuar?')) return;
      if(backup.config){CONFIG=backup.config;saveConfig(CONFIG);}
      if(backup.state) state=backup.state;
      state.driveRevision=Number(backup.revision||(backup.state&&backup.state.driveRevision)||0);
      state.driveLastSyncISO=todayISO(); saveState(state); setBaseline(CONFIG,state,state.driveRevision);
      if(typeof render==='function') render();
      alert('Dados carregados e revisão sincronizada!');
    }catch(err){console.error('Drive load error',err);alert('Não consegui carregar do Google Drive. Confira a implantação do Apps Script.');}
  };

  setTimeout(()=>{const b=getBaseline();if(!b&&typeof state!=='undefined'&&state&&Number(state.driveRevision||0)>0)setBaseline(CONFIG,state,state.driveRevision);},0);
})();
</script>
